\begin{epigraph}
\textit{``If we use, to achieve our purposes, a mechanical agency with whose operation we cannot efficiently interfere once we have started it\ldots\ we had better be quite sure that the purpose put into the machine is the purpose which we really desire and not merely a colorful imitation of it.''}\par
\vspace{1em}
\normalfont Norbert Wiener, 1960\footnote{Norbert Wiener, ``Some Moral and Technical Consequences of Automation,'' \textit{Science} 131, no. 3410 (1960): 1355, https://doi.org/10.1126/science.131.3410.1355.}
\end{epigraph}
